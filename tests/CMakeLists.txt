# =============================================================================
# Build-system regression checks (static — no toolchain/configure needed).
# Registered as a ctest test; runs the cmake_invariants.cmake script which
# asserts the build-system invariants from the CMake audit.
# =============================================================================
add_test(NAME build_invariants
    COMMAND "${CMAKE_COMMAND}"
            -DREPO=${CMAKE_SOURCE_DIR}
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake_invariants.cmake")
set_tests_properties(build_invariants PROPERTIES
    FAIL_REGULAR_EXPRESSION "FAIL;regressed")

# =============================================================================
# metal_kernels_compile (macOS with Neurosuite::metal): every *.metal under
# src/ embedded and compiled at run time, with a pipeline for each kernel --
# the check that Apple's compiler accepts all of them, not only those of the
# programs the numerical tests run.  SKIPPED without a Metal device.
# =============================================================================
if(TARGET Neurosuite::metal)
    include("${CMAKE_SOURCE_DIR}/cmake/MetalBackend.cmake")   # ns_metal_embed
    file(GLOB_RECURSE _ns_metal_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.metal")
    list(SORT _ns_metal_sources)
    add_executable(metal_kernels_compile metal_kernels_compile.cpp)
    target_link_libraries(metal_kernels_compile PRIVATE Neurosuite::metal)
    set(_decls "")
    set(_rows "")
    set(_i 0)
    foreach(_src IN LISTS _ns_metal_sources)
        file(RELATIVE_PATH _rel "${CMAKE_SOURCE_DIR}" "${_src}")
        ns_metal_embed(metal_kernels_compile ns_metal_src_${_i} "${_src}")
        string(APPEND _decls "extern const char ns_metal_src_${_i}[];\n")
        string(APPEND _rows "    { \"${_rel}\", ns_metal_src_${_i} },\n")
        math(EXPR _i "${_i} + 1")
    endforeach()
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/metal_sources_table.cpp"
        CONTENT "// Generated by tests/CMakeLists.txt; do not edit.\nstruct MetalSourceEntry { const char* path; const char* source; };\n${_decls}extern const MetalSourceEntry kMetalSources[];\nextern const int kMetalSourceCount;\nconst MetalSourceEntry kMetalSources[] = {\n${_rows}};\nconst int kMetalSourceCount = ${_i};\n")
    target_sources(metal_kernels_compile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/metal_sources_table.cpp")
    add_test(NAME metal_kernels_compile COMMAND metal_kernels_compile)
    set_tests_properties(metal_kernels_compile PROPERTIES SKIP_REGULAR_EXPRESSION "SKIPPED")
endif()
